**Goal:** The simulation service runs Starsim 3.7.2 through a Python builder that reproduces the v1 template behaviour, and no Jinja template is on the `/simulate` path.

**Architecture:** A new package `epichat/sim/` holds the six model classes (`models.py`), the builder (`builder.py`), the result extraction (`outputs.py`) and a subprocess entry point (`run_params.py`). The service's runner spawns `python -m epichat.sim.run_params <params.json> <out.json>` instead of a rendered script; the repair loop, timeout, cap and response assembly are unchanged. A parity baseline captured on 3.3.2 with the template pipeline, before any pin changes, pins the behaviour: medians over ten seeds per model within 5% (total infected) and 3 days (peak day).

**Tech Stack:** Python 3.12, Starsim 3.7.2, pydantic 2, FastAPI `TestClient`, pytest. No web code changes beyond a `package.json` script line, a DEPLOY.md revision and one deploy-doc pin test.

**Spec:** `docs/superpowers/specs/2026-10-09-sim-parameters-disease-design.md`, sections 2, 3, 7.1, 8 (as revised by section 17), 12 (phase 0), 13 (parity) and 17.

```python
"""Capture the v1 parity baseline (spec section 13): each model's default
parameters at 10k agents for one year, seeds 1 to 10, run through the Jinja
templates on the Starsim version installed in the interpreter that runs this
script. Run it ONCE with the service venv while it still has Starsim 3.3.2:

    sim/.venv/Scripts/python.exe scripts/capture_sim_baseline.py

It writes tests/fixtures/sim_baseline_v1.json. Rerunning on a newer Starsim
would not be a baseline; the test pins the version."""
from __future__ import annotations

import json
import statistics
import sys
import tempfile
import uuid
from importlib.metadata import version
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
for path in (str(ROOT), str(ROOT / "sim")):
    if path not in sys.path:
        sys.path.insert(0, path)

from epichat.generator import CodeGenerator  # noqa: E402
from epichat.schema import SimParams  # noqa: E402

BASE = dict(beta=22.8125, n_agents=10000, sim_dur_years=1.0, n_contacts=4, init_prev=0.01, dur_inf=10.0, p_death=0.0)
CASES = {
    "sir": dict(disease_type="sir"),
    "sis": dict(disease_type="sis"),
    "sirs": dict(disease_type="sirs", dur_immune=180.0),
    "seir": dict(disease_type="seir", dur_exp=5.0),
    "seirs": dict(disease_type="seirs", dur_exp=5.0, dur_immune=180.0),
    "seiar": dict(disease_type="seiar", dur_exp=5.0, p_asymp=0.3, rel_trans_asymp=0.5),
}
SEEDS = list(range(1, 11))
OUT = ROOT / "tests" / "fixtures" / "sim_baseline_v1.json"


def case_params(name: str) -> dict:
    return {**BASE, **CASES[name]}


def run_one(params: SimParams) -> dict:
    from runner import run_script  # sim/runner.py: the 3.3.2 subprocess runner
    out = Path(tempfile.gettempdir()) / f"epichat-baseline-{uuid.uuid4().hex}.json"
    code = CodeGenerator().generate(params, str(out), output_mode="json")
    result = run_script(code, out, timeout=600)
    if not result.ok:
        raise RuntimeError(f"{params.disease_type} seed {params.rand_seed}: {result.error}")
    return result.stats


def main() -> None:
    cases = {}
    for name in CASES:
        per_seed = []
        for seed in SEEDS:
            stats = run_one(SimParams(**case_params(name), rand_seed=seed))
            per_seed.append({"seed": seed, "total_infected": int(stats["total_infected"]),
                             "peak_day": int(stats["peak_day"]), "peak_infections": int(stats["peak_infections"])})
            print(name, seed, per_seed[-1], flush=True)
        cases[name] = {
            "params": case_params(name),
            "per_seed": per_seed,
            "median": {key: statistics.median(r[key] for r in per_seed) for key in ("total_infected", "peak_day")},
        }
    OUT.write_text(json.dumps({
        "starsim_version": version("starsim"), "n_agents": BASE["n_agents"], "sim_dur_years": BASE["sim_dur_years"],
        "seeds": SEEDS, "cases": cases,
    }, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    print("wrote", OUT)


if __name__ == "__main__":
    main()
```

```python
import pytest
import starsim as ss

from epichat.sim.models import MODELS, SEIAR, SEIR, SEIRS, SIR, SIRS, SIS


def run(model, days=40, n=800, seed=1, **pars):
    sim = ss.Sim(n_agents=n, networks=ss.RandomNet(n_contacts=4), diseases=model(**pars),
                 start=ss.days(0), dur=ss.days(days), dt=ss.days(1), rand_seed=seed, verbose=0)

    assert MODELS == {"sir": SIR, "seir": SEIR, "sis": SIS, "sirs": SIRS, "seirs": SEIRS, "seiar": SEIAR}


@pytest.mark.parametrize("name", ["sir", "seir", "sirs", "seirs", "seiar"])
def test_models_count_the_deaths_they_request(name):
    pars = dict(init_prev=ss.bernoulli(p=0.05), beta=ss.peryear(30), dur_inf=ss.constant(ss.days(5)), p_death=ss.bernoulli(p=0.5))
    if name in ("seir", "seirs", "seiar"):
        pars["dur_exp"] = ss.constant(ss.days(2))
    if name in ("sirs", "seirs"):
        pars["dur_immune"] = ss.constant(ss.days(30))
    sim = run(MODELS[name], **pars)
    disease_deaths = sim.results[name].new_deaths_disease.values
    assert disease_deaths.sum() > 0
    # no demographics, so every death in the population is a disease death
    assert disease_deaths.sum() == sim.results.cum_deaths.values[-1]
    assert sim.results[name].timevec is not None and len(disease_deaths) == 41


def test_sis_has_no_immunity_and_no_deaths_by_default():
    sim = run(SIS, init_prev=ss.bernoulli(p=0.05), beta=ss.peryear(30), dur_inf=ss.constant(ss.days(5)),

    assert np.all(d.rel_sus.values == 1.0)
    res = sim.results.sis
    assert "n_recovered" not in res and "new_deaths_disease" in res
    assert res.new_deaths_disease.values.sum() == 0
    assert res.n_susceptible.values[-1] + res.n_infected.values[-1] == 800
```

```python
"""EpiChat's six compartmental models as thin subclasses of Starsim 3.7's
SIR, SEIR and SIS (spec 7.1).

Phase 0 reproduces the v1 templates: constant durations arrive from the
builder, deaths are drawn for every infection (seiar included), asymptomatic
agents lower rel_trans directly, and sis has neither immunity (the builder
passes waning 0 and imm_boost 0) nor deaths. The one addition is the
`new_deaths_disease` result, which counts the deaths a disease requests so the
service no longer subtracts background mortality.
"""
from __future__ import annotations

import numpy as np
import starsim as ss


class EpiChatDisease:
    """Mixin for every EpiChat model: counts the deaths this disease requests."""

    def init_results(self):
        super().init_results()
        self.define_results(ss.Result("new_deaths_disease", dtype=int, scale=True, label="Disease deaths"))

    def step_state(self):
        if hasattr(self, "ti_dead"):   # ss.SIS has no deaths
            deaths = (self.ti_dead <= self.ti).uids
            if len(deaths):
                self.results.new_deaths_disease[self.ti] += len(deaths)
        super().step_state()


class SIR(EpiChatDisease, ss.SIR):
    pass


class SEIR(EpiChatDisease, ss.SEIR):
    pass


class SIS(EpiChatDisease, ss.SIS):
    pass


class _Waning:
    """R -> S after dur_immune, for SIRS and SEIRS. `start` is the state the
    infectious period is counted from (ti_infected in SIR, ti_infectious in SEIR)."""

    start_state = "ti_infected"

    def _define_waning(self, dur_immune):
        self.define_pars(dur_immune=ss.constant(ss.days(180)))
        self.update_pars(dur_immune=dur_immune)
        self.define_states(ss.FloatArr("ti_susceptible", label="Time immunity wanes"))

    def set_progression(self, uids):
        p = self.pars
        start = getattr(self, self.start_state)
        dur_inf = p.dur_inf.rvs(uids)
        dur_imm = p.dur_immune.rvs(uids)
        will_die = p.p_death.rvs(uids)
        dead, rec = uids[will_die], uids[~will_die]
        self.ti_dead[dead] = start[dead] + dur_inf[will_die]
        self.ti_recovered[dead] = np.nan
        self.ti_recovered[rec] = start[rec] + dur_inf[~will_die]
        self.ti_dead[rec] = np.nan
        self.ti_susceptible[rec] = self.ti_recovered[rec] + dur_imm[~will_die]

    def step_state(self):
        super().step_state()
        waned = (self.recovered & (self.ti_susceptible <= self.ti)).uids
        self.recovered[waned] = False
        self.susceptible[waned] = True


class SIRS(_Waning, SIR):
    def __init__(self, pars=None, dur_immune=None, **kwargs):
        super().__init__(pars=pars, **kwargs)
        self._define_waning(dur_immune)


class SEIRS(_Waning, SEIR):
    start_state = "ti_infectious"

    def __init__(self, pars=None, dur_immune=None, **kwargs):
        super().__init__(pars=pars, **kwargs)
        self._define_waning(dur_immune)


class SEIAR(SEIR):
    """SEIR with an asymptomatic track: E -> I or A -> R; A transmits at rel_trans_asymp."""

    def __init__(self, pars=None, p_asymp=None, rel_trans_asymp=None, **kwargs):
        super().__init__(pars=pars, **kwargs)
        self.define_pars(p_asymp=ss.bernoulli(p=0.3), rel_trans_asymp=0.5)
        self.update_pars(p_asymp=p_asymp, rel_trans_asymp=rel_trans_asymp)
        self.define_states(
            ss.BoolState("asymptomatic", label="Asymptomatic"),
            ss.BoolArr("asymp_track", label="Will be asymptomatic once infectious"),
        )

    def set_progression(self, uids):
        super().set_progression(uids)                 # phase 0: deaths drawn for every infection, as v1
        self.asymp_track[uids] = self.pars.p_asymp.rvs(uids)

    def step_state(self):
        becoming = (self.exposed & (self.ti_infectious <= self.ti)).uids
        super().step_state()
        asymp = becoming[self.asymp_track[becoming]]
        self.asymptomatic[asymp] = True
        self.rel_trans[asymp] = self.pars.rel_trans_asymp   # phase 0: direct, as v1; phase 2 moves to rel_trans_base

    def clear_infection(self, uids):
        super().clear_infection(uids)
        self.asymptomatic[uids] = False
        self.asymp_track[uids] = False
        self.rel_trans[uids] = 1.0


MODELS: dict[str, type] = {"sir": SIR, "seir": SEIR, "sis": SIS, "sirs": SIRS, "seirs": SEIRS, "seiar": SEIAR}
```

```python
"""build_sim: a validated v1 SimParams -> an initialised Starsim 3.7 Sim with
the behaviour of the v1 templates (spec 8, as revised by 17).

Time is numeric and day-based (start 0, dt one day, dur round(365 × years)
days), which gives v1's point count. Every duration is ss.days and every rate
ss.peryear. The carried-over pieces (treatment, routine vaccine, the day-0
vaccine move, age assignment, demographics, seasonality) keep their v1
semantics; the day-0 move and the age assignment keep v1's numpy generators
until phase 2 replaces them.
"""
from __future__ import annotations

import numpy as np
import starsim as ss

from epichat.schema import SimParams

from .models import MODELS

CONTACT_MATRIX = np.array([[7.0, 2.5, 0.5], [2.5, 9.0, 1.5], [0.5, 1.5, 3.5]])
LATENT_MODELS = ("seir", "seirs", "seiar")
WANING_MODELS = ("sirs", "seirs")


def sim_days(sim_dur_years: float) -> int:
    return max(1, int(round(365 * sim_dur_years)))


def resolve_seed(params: SimParams, seed: int | None) -> int | None:
    return seed if seed is not None else params.rand_seed


def age_groups() -> dict:
    return {"children": ss.AgeGroup(0, 18), "adults": ss.AgeGroup(18, 65), "elderly": ss.AgeGroup(65, None)}


class SimpleTreatment(ss.Intervention):
    """v1's _SimpleTreatment: each day from start_day, up to prob × (infectious agents),
    capped at max_capacity, recover on the next step."""

    def __init__(self, prob=1.0, max_capacity=None, start_day=0):
        super().__init__()
        self.prob = prob
        self.max_capacity = max_capacity
        self.start_day = start_day

    def step(self):
        if self.ti < self.start_day:
            return
        disease = self.sim.diseases[0]
        infectious = disease.infectious.uids
        if len(infectious) == 0:
            return
        n = int(self.prob * len(infectious))
        if self.max_capacity is not None:
            n = min(n, self.max_capacity)
        if n <= 0:
            return
        rng = np.random.default_rng(int(self.ti))
        treated = ss.uids(rng.choice(np.asarray(infectious), size=min(n, len(infectious)), replace=False))
        disease.ti_recovered[treated] = self.ti + 1


def disease_kwargs(params: SimParams) -> dict:
    kw = dict(init_prev=ss.bernoulli(p=params.init_prev), beta=ss.peryear(params.beta),
              dur_inf=ss.constant(ss.days(params.dur_inf)))
    if params.disease_type == "sis":
        kw.update(waning=ss.peryear(0), imm_boost=0)       # v1 sis: no immunity, no deaths
    else:
        kw["p_death"] = ss.bernoulli(p=params.p_death)
    if params.disease_type in LATENT_MODELS:
        kw["dur_exp"] = ss.constant(ss.days(params.dur_exp))
    if params.disease_type in WANING_MODELS:
        kw["dur_immune"] = ss.constant(ss.days(params.dur_immune))
    if params.disease_type == "seiar":
        kw.update(p_asymp=ss.bernoulli(p=params.p_asymp), rel_trans_asymp=params.rel_trans_asymp)
    return kw


def build_network(params: SimParams):
    if params.network_type == "age_structured":
        return ss.MixingPools(src=age_groups(), dst=age_groups(), n_contacts=CONTACT_MATRIX * params.network_beta, beta=1.0)
    return ss.RandomNet(n_contacts=params.n_contacts, beta=params.network_beta)


def _move_to_immune(sim: ss.Sim, coverage: float, seed: int | None) -> None:
    """v1's day-0 vaccine: coverage × susceptibles become immune before the run."""
    disease = sim.diseases[0]
    susceptible = np.asarray(disease.susceptible.uids)
    n_vax = int(coverage * len(susceptible))
    rng = np.random.default_rng(seed) if seed is not None else np.random.default_rng()
    chosen = ss.uids(rng.choice(susceptible, size=n_vax, replace=False)) if n_vax > 0 else ss.uids()
    if hasattr(disease, "recovered"):
        disease.susceptible[chosen] = False
        disease.recovered[chosen] = True
    else:                                               # sis: immune through full immunity (waning is 0)
        disease.immunity[chosen] = 1.0
        disease.rel_sus[chosen] = 0.0


def _assign_ages(sim: ss.Sim, params: SimParams, seed: int | None) -> None:
    """v1's age assignment: uniform ages within each share, shuffled."""
    n = len(sim.people)
    n_children = round(params.age_pct_under18 / 100 * n)
    n_elderly = round(params.age_pct_over65 / 100 * n)
    n_adults = max(0, n - n_children - n_elderly)
    rng = np.random.default_rng(seed + 1) if seed is not None else np.random.default_rng()
    ages = np.concatenate([rng.uniform(0, 18, n_children), rng.uniform(18, 65, n_adults), rng.uniform(65, 95, n_elderly)])
    rng.shuffle(ages)
    sim.people.age[:] = ages[:n]


def build_sim(params: SimParams, seed: int | None = None) -> ss.Sim:
    name = params.disease_type
    disease = MODELS[name](**disease_kwargs(params))
    interventions, connectors = [], []
    seasonality = params.get_seasonality()
    if seasonality:
        connectors.append(ss.seasonality(diseases=name, scale=seasonality.scale, shift=seasonality.shift))
    treatment = params.get_treatment()
    if treatment:
        interventions.append(SimpleTreatment(prob=treatment.coverage if treatment.coverage is not None else 1.0,
                                             max_capacity=treatment.capacity, start_day=treatment.start_day))
    vaccine = params.get_vaccine()
    if vaccine and vaccine.start_day > 0:
        interventions.append(ss.routine_vx(product=ss.simple_vx(efficacy=1.0), prob=vaccine.coverage or 0.0,
                                           start_year=vaccine.start_day / 365))
    demographics = None
    if params.use_demographics:
        demographics = [ss.Births(birth_rate=ss.peryear(params.birth_rate)), ss.Deaths(death_rate=ss.peryear(params.death_rate))]
    rand_seed = resolve_seed(params, seed)
    seed_kw = {"rand_seed": rand_seed} if rand_seed is not None else {}
    sim = ss.Sim(n_agents=params.n_agents, networks=build_network(params), diseases=disease, demographics=demographics,
                 interventions=interventions or None, connectors=connectors or None,
                 start=ss.days(0), dur=ss.days(sim_days(params.sim_dur_years)), dt=ss.days(1), verbose=0, **seed_kw)
    sim.init()
    if vaccine and vaccine.start_day <= 0:
        _move_to_immune(sim, vaccine.coverage or 0.0, rand_seed)
    if params.network_type == "age_structured" and params.age_pct_under18 is not None:
        _assign_ages(sim, params, rand_seed)
    return sim
```

```python
"""Series and stats from a finished sim, in the v1 shape the service, the web
app and the report already consume (spec 17; the 8.6 key set is phase 2)."""
from __future__ import annotations

import numpy as np
import starsim as ss

V1_SERIES_KEYS = ("n_susceptible", "n_exposed", "n_infected", "n_asymptomatic", "n_recovered", "new_infections", "cum_infections")


def extract(sim: ss.Sim, n_agents: int) -> dict:
    name = list(sim.diseases.keys())[0]
    res = sim.results[name]
    npts = len(sim.results.timevec)
    infectious_key = "n_infectious" if "n_infectious" in res else "n_infected"
    series: dict[str, list] = {"day": list(range(npts))}
    for key in V1_SERIES_KEYS:
        source = infectious_key if key == "n_infected" else key
        if source in res:
            series[key] = [float(v) for v in res[source].values]
    deaths = np.asarray(res["new_deaths_disease"].values, dtype=float) if "new_deaths_disease" in res else np.zeros(npts)
    series["new_deaths"] = [float(v) for v in deaths]
    series["cum_deaths"] = [float(v) for v in np.cumsum(deaths)]
    n_infected = np.asarray(series["n_infected"])
    stats = {
        "peak_infections": int(n_infected.max()),
        "peak_day": int(n_infected.argmax()),
        "total_infected": int(series["cum_infections"][-1]),
        "total_deaths": int(series["cum_deaths"][-1]),
        "n_agents": int(n_agents),
        "sim_days": npts,
    }
    return {"stats": stats, "series": series}
```

```python
import json
import subprocess
import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parent.parent
SMALL = {"disease_type": "sir", "beta": 22.8125, "n_agents": 300, "sim_dur_years": 10 / 365, "rand_seed": 1}


def run_child(params: dict, tmp_path: Path, args=None):
    params_path = tmp_path / "params.json"
    out_path = tmp_path / "out.json"
    params_path.write_text(json.dumps(params), encoding="utf-8")
    cmd = [sys.executable, "-m", "epichat.sim.run_params"] + ([str(params_path), str(out_path)] if args is None else args)
    completed = subprocess.run(cmd, capture_output=True, text=True, encoding="utf-8", errors="replace", cwd=ROOT, timeout=300)
    return completed, out_path
```

```python
"""Subprocess entry point for the simulation service (spec 8.8):

    python -m epichat.sim.run_params <params.json> <out.json>

The parent writes the validated parameters as JSON, this process imports
Starsim (the expensive part), builds and runs the sim, and writes
{"stats", "series"} to the output path. Any failure is a traceback on stderr
and a non-zero exit; the parent keeps the stderr tail as the error."""
from __future__ import annotations

import json
import sys

from epichat.schema import SimParams

from .builder import build_sim
from .outputs import extract


def run(params: SimParams) -> dict:
    sim = build_sim(params)
    sim.run()
    return extract(sim, params.n_agents)


def main(argv: list[str] | None = None) -> int:
    argv = sys.argv[1:] if argv is None else argv
    if len(argv) != 2:
        print("usage: python -m epichat.sim.run_params <params.json> <out.json>", file=sys.stderr)
        return 2
    with open(argv[0], encoding="utf-8") as f:
        params = SimParams(**json.load(f))
    payload = run(params)
    with open(argv[1], "w", encoding="utf-8") as f:
        json.dump(payload, f)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

Replace `sim/runner.py` with:

```python
"""Run one simulation in a child process.

The child is `python -m epichat.sim.run_params <params.json> <out.json>`: it
imports Starsim, builds the sim from the validated parameters and writes the
result file. The parent passes its import path (so the copied package and the
installed Starsim are found on Vercel), writable numba and matplotlib cache
directories, and a hard timeout. stdout is ignored; stderr's tail is the error."""
from __future__ import annotations

import json
import os
import subprocess
import sys
import tempfile
import time
from dataclasses import dataclass
from pathlib import Path

from epichat.schema import SimParams

STDERR_TAIL = 2000


@dataclass
class RunResult:
    ok: bool
    stats: dict
    series: dict
    error: str | None
    timed_out: bool
    duration_ms: int


def child_environment() -> dict[str, str]:
    env = dict(os.environ)
    env["PYTHONPATH"] = os.pathsep.join(p for p in sys.path if p)
    scratch = Path(tempfile.gettempdir()) / "epichat-sim"
    env["NUMBA_CACHE_DIR"] = str(scratch / "numba")
    env["MPLCONFIGDIR"] = str(scratch / "mpl")
    env.setdefault("PYTHONIOENCODING", "utf-8")
    return env


def run_params(params: SimParams, output_path: Path, timeout: float) -> RunResult:
    output_path = Path(output_path)
    started = time.perf_counter()
    with tempfile.NamedTemporaryFile("w", prefix="epichat-params-", suffix=".json", delete=False, encoding="utf-8") as f:
        json.dump(params.model_dump(mode="json"), f)
        params_path = Path(f.name)

    def elapsed() -> int:
        return int((time.perf_counter() - started) * 1000)

    try:
        completed = subprocess.run(
            [sys.executable, "-m", "epichat.sim.run_params", str(params_path), str(output_path)],
            capture_output=True, text=True, encoding="utf-8", errors="replace",
            timeout=timeout, env=child_environment(),
        )
    except subprocess.TimeoutExpired:
        output_path.unlink(missing_ok=True)
        return RunResult(False, {}, {}, f"Simulation timed out after {timeout:g} seconds.", True, elapsed())
    finally:
        params_path.unlink(missing_ok=True)

    try:
        if completed.returncode != 0:
            tail = (completed.stderr or "").strip()[-STDERR_TAIL:] or f"exit code {completed.returncode}"
            return RunResult(False, {}, {}, tail, False, elapsed())
        if not output_path.exists():
            return RunResult(False, {}, {}, "Simulation finished but wrote no result file.", False, elapsed())
        with open(output_path, encoding="utf-8") as f:
            payload = json.load(f)
        return RunResult(True, payload.get("stats", {}), payload.get("series", {}), None, False, elapsed())
    finally:
        output_path.unlink(missing_ok=True)
```
